// Core Equations — ASHRAE-style Pool Surface Load
pws(T_°F) = exp(14.43509 − 8162.67 / (T + 459.67)) // sat. pressure [psia]
Evap [lb/hr] = A_ft² × (pw − pa)_inHg × (0.1 + 0.015 × V_fpm)
Q_latent [BTU/hr] = Evap × 1050
h_conv = 0.22 + 0.15 × √V_fpm
Q_sensible [BTU/hr] = h_conv × A_ft² × max(0, T_w − T_a)
// Outdoor Air Penalty
w = 0.62198 × p_v / (14.696 − p_v) // humidity ratio [lb/lb]
Q_OA,s = 1.08 × CFM × max(0, T_out − T_in)
Q_OA,l = 4840 × CFM × max(0, w_out − w_in)
